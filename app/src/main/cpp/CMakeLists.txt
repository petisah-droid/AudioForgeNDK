cmake_minimum_required(VERSION 3.22.1)
project("audioforge_ndk" LANGUAGES CXX)
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/third_party/install/include)
add_library(mp3lame STATIC IMPORTED)
set_target_properties(mp3lame PROPERTIES IMPORTED_LOCATION ${CMAKE_CURRENT_SOURCE_DIR}/third_party/install/lib/libmp3lame.a)
add_library(flac STATIC IMPORTED)
set_target_properties(flac PROPERTIES IMPORTED_LOCATION ${CMAKE_CURRENT_SOURCE_DIR}/third_party/install/lib/libFLAC.a)
add_library(audioforge_ndk SHARED audio_engine.cpp)
find_library(log-lib log)
find_library(aaudio-lib aaudio)
find_library(mediandk-lib mediandk)
target_link_libraries(audioforge_ndk flac mp3lame ${log-lib} ${aaudio-lib} ${mediandk-lib})
